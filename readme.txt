SELECT
    COALESCE(t.COLUMN_NAME, v.COLUMN_NAME) AS ColumnName,
    CASE
        WHEN t.COLUMN_NAME IS NOT NULL
         AND v.COLUMN_NAME IS NOT NULL THEN 'Matched'
        WHEN t.COLUMN_NAME IS NOT NULL THEN 'Only in Table'
        ELSE 'Only in View'
    END AS Status
FROM
    (SELECT COLUMN_NAME
     FROM INFORMATION_SCHEMA.COLUMNS
     WHERE TABLE_SCHEMA = 'dbo'
       AND TABLE_NAME = 'AccountSummary') t
FULL OUTER JOIN
    (SELECT COLUMN_NAME
     FROM INFORMATION_SCHEMA.COLUMNS
     WHERE TABLE_SCHEMA = 'dbo'
       AND TABLE_NAME = 'vw_AccountSummary') v
ON t.COLUMN_NAME = v.COLUMN_NAME
ORDER BY ColumnName;
